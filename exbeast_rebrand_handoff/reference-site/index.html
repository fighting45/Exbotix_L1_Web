hardhat.config.js
// ExBeast Mainnet · 11311  —  Testnet · 11211
module.exports = {
  networks: {
    exbotix: {           // Mainnet
      url: "https://rpc.exbotix.net/",
      chainId: 11311,
      accounts: [process.env.PRIVATE_KEY],
    },
    exbotixTestnet: {    // Testnet
      url: "http://testnet.rpc.exbotix.net",
      chainId: 11211,
      accounts: [process.env.PRIVATE_KEY],
    },
  },
};

// Deploy as usual
$ npx hardhat run deploy.js --network exbotix
